SELECT Applications.appID, 
       Applications.applicationName, 
       Applications.applicationType, 
       Applications.applicationStatus, 
       Entities.entityName, 
       Applications.portalAppID, 
       Applications.initialScore, 
       Applications.fundingRequest, 
       Entities.entityType, 
       Entities.entityID
FROM Applications 
    INNER JOIN Entities ON Applications.entityID = Entities.entityID
WHERE entityType = "Applicant"
    AND Applications.entityID = '{{Select9.selectedOptionValue}}'
ORDER BY entityName ASC, applicationName ASC;
