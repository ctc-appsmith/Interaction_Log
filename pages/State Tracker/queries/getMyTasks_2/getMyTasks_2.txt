SELECT 
    Tasks.taskID, 
    Tasks.appID, 
    Tasks.startDate, 
    Tasks.completedDate, 
    Tasks.dueDate, 
    Tasks.taskTypeID, 
    Tasks.taskStatus, 
    Tasks.assignedTo, 
    Tasks.taskNotes, 
    Tasks.filePath, 
    Applications.applicationName, 
    Applications.portalAppID, 
    Applications.applicationType, 
    Applications.reviewOrder, 
    Entities.entityName,
		Entities.entityID,
    taskTypes.taskName, 
    taskTypes.team, 
    taskStatusTypes.sortOrder
FROM taskStatusTypes 
INNER JOIN (((Tasks 
    INNER JOIN Applications ON Tasks.appID = Applications.appID) 
    INNER JOIN Entities ON Applications.entityID = Entities.entityID) 
    INNER JOIN taskTypes ON Tasks.taskTypeID = taskTypes.taskTypeID) 
    ON taskStatusTypes.taskStatus = Tasks.taskStatus
WHERE Entities.clientID = {{appsmith.store.clientID}}
		AND Applications.applicationStatus <> "Ineligible"
		AND Applications.applicationStatus <> "Withdrawn"
    AND ({{Switch1.isSwitchedOn}} OR Tasks.taskStatus <> "Complete")
		AND (({{!Select13.selectedOptionValue}} OR (Entities.entityID = '{{Select13.selectedOptionValue}}')))
		AND (({{!Select6Copy.selectedOptionValue}} OR (Tasks.appID = '{{Select6Copy.selectedOptionValue}}')))
    AND ({{!Select7Copy1.selectedOptionValue}} OR (taskTypes.team = '{{Select7Copy1.selectedOptionValue}}'))
		AND ({{!Select7Copy1Copy.selectedOptionValue}} OR (Tasks.assignedTo = '{{Select7Copy1Copy.selectedOptionValue}}'))
ORDER BY reviewOrder ASC, sortOrder ASC, startDate DESC;
