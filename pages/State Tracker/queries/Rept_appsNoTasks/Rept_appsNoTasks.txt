SELECT 
    Applications.appID, 
    Applications.entityID, 
    Applications.applicationName, 
    Applications.portalAppID, 
    Applications.applicationStatus
FROM 
    Applications 
    LEFT JOIN Entities ON Applications.entityID = Entities.entityID 
    LEFT JOIN Clients ON Entities.clientID = Clients.clientID
WHERE 
    (Clients.clientID = {{appsmith.store.clientID}})
    AND Applications.applicationStatus NOT IN ('Approved', 'Rejected', 'Withdrawn', 'Ineligible')
    AND NOT EXISTS (
        SELECT 1 
        FROM Tasks 
        WHERE Tasks.appID = Applications.appID
        AND Tasks.taskStatus != 'Complete'
    );
