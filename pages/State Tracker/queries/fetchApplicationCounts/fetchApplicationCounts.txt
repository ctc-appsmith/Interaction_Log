SELECT 
    Applications.applicationStatus, 
    Applications.reviewOrder,
    COUNT(Applications.appID) AS CountOfappID
FROM 
    Applications 
    INNER JOIN Entities ON Applications.entityID = Entities.entityID
    INNER JOIN Clients ON Entities.clientID = Clients.clientID
WHERE 
    Clients.clientID = {{appsmith.store.clientID}}
GROUP BY 
    Applications.applicationStatus, Applications.reviewOrder
ORDER BY 
    Applications.applicationStatus, CountOfappID DESC;
