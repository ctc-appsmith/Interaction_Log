SELECT taskTypes.team, Tasks.completedDate, Count(Tasks.taskID) AS CountOftaskID
FROM ((Clients INNER JOIN Entities ON Clients.clientID = Entities.clientID) INNER JOIN (Tasks INNER JOIN Applications ON Tasks.appID = Applications.appID) ON Entities.entityID = Applications.entityID) INNER JOIN taskTypes ON Tasks.taskTypeID = taskTypes.taskTypeID
WHERE (((Clients.clientID)={{appsmith.store.clientID}}) AND ((Tasks.taskStatus)="Complete"))
GROUP BY taskTypes.team, Tasks.completedDate
ORDER BY taskTypes.team;
