SELECT 
	Count(Tasks.taskID) AS CountOftaskID
		FROM ((Tasks INNER JOIN Applications ON Tasks.appID = Applications.appID) 
					INNER JOIN Entities ON Applications.entityID = Entities.entityID) 
					INNER JOIN Clients ON Entities.clientID = Clients.clientID


WHERE 
    Tasks.taskStatus <> 'Complete'
		AND Tasks.taskStatus<>'Application Ineligible'
		AND Tasks.taskStatus<>'Application Withdrawn'
    AND Clients.clientID = {{appsmith.store.clientID}}
;
