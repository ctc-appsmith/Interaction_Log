SELECT
  appStatusID,
	appStatusType,
	statusDefinition,
	processOrder,
	clientID
FROM
  applicationStatusTypes
	
	WHERE clientID = {{appsmith.store.clientID}}

ORDER BY
  {{appStatusType_Table.sortOrder.column || 'processOrder'}} 
LIMIT
  {{appStatusType_Table.pageSize}}
OFFSET
  {{appStatusType_Table.pageOffset}}
