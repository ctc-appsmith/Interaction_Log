SELECT Count(Tasks.taskID) AS CountOftaskID
FROM ((Tasks 
    INNER JOIN Applications ON Tasks.appID = Applications.appID) 
    INNER JOIN Entities ON Applications.entityID = Entities.entityID) 
    INNER JOIN taskTypes ON Tasks.taskTypeID = taskTypes.taskTypeID
WHERE 
    Tasks.taskStatus <> 'Complete' 
    AND Entities.clientID = {{appsmith.store.clientID}}
		AND Applications.applicationStatus <> "Ineligible"
		AND Applications.applicationStatus <> "Withdrawn"
    AND (
        -- If Select11 is null, include all application types
        ({{!Select11.selectedOptionValue}} OR Applications.applicationType = '{{Select11.selectedOptionLabel}}')
    )
    AND (
        -- If Select7Copy is null, include all teams
        ({{!Select7Copy.selectedOptionValue}} OR taskTypes.team = '{{Select7Copy.selectedOptionValue}}')
    );
