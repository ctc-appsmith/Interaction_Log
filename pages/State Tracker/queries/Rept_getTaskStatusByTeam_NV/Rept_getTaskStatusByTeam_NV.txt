SELECT 
    taskTypes.team,
    Applications.reviewOrder AS Priority,
    COUNT(Tasks.taskID) AS CountOftaskID
FROM Clients 
INNER JOIN Entities ON Clients.clientID = Entities.clientID
INNER JOIN Applications ON Applications.entityID = Entities.entityID
INNER JOIN Tasks ON Tasks.appID = Applications.appID
INNER JOIN taskTypes ON Tasks.taskTypeID = taskTypes.taskTypeID
WHERE 
    Tasks.taskStatus NOT IN ('Complete', 'Application Ineligible', 'Application Withdrawn')
    AND Clients.clientID = {{appsmith.store.clientID}}
GROUP BY 
    taskTypes.team, 
    Priority
ORDER BY 
    taskTypes.team, 
    Priority;
