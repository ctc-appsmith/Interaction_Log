SELECT 
	taskTypes.team,
	Applications.reviewOrder AS Priority,
	Count(Tasks.taskID) AS CountOftaskID
	FROM Clients 
		INNER JOIN (((Tasks INNER JOIN taskTypes ON Tasks.taskTypeID = taskTypes.taskTypeID) 
								 INNER JOIN Applications ON Tasks.appID = Applications.appID) 
								INNER JOIN Entities ON Applications.entityID = Entities.entityID) ON Clients.clientID = Entities.clientID
	WHERE (((Tasks.taskStatus)<>"Complete") AND ((Clients.clientID)={{appsmith.store.clientID}}))
GROUP BY taskTypes.team, Priority
ORDER BY taskTypes.team, Priority;
