SELECT 
    Applications.techType,
    Applications.reviewOrder,
    COUNT(Applications.appID) AS CountOfappID
	FROM 
		(Applications INNER JOIN Entities ON Applications.entityID = Entities.entityID) 
	WHERE Applications.applicationType = 'Scoring'
	AND (((Entities.clientID)={{appsmith.store.clientID}}))
GROUP BY Applications.techType, Applications.reviewOrder
ORDER BY Applications.techType, Applications.reviewOrder;
