SELECT 
	taskTypes.team, 
	Applications.reviewOrder AS Priority,
	COUNT(Tasks.taskID) AS CountOftaskID
	FROM 
		Clients INNER JOIN (((Tasks INNER JOIN taskTypes ON Tasks.taskTypeID = taskTypes.taskTypeID) 
												 INNER JOIN Applications ON Tasks.appID = Applications.appID) 
												INNER JOIN Entities ON Applications.entityID = Entities.entityID) ON Clients.clientID = Entities.clientID
	WHERE (((Tasks.taskStatus)<>"Complete") AND ((Tasks.assignedTo)="unassigned@ctcnet.us") 
				 AND ((Clients.clientID)={{appsmith.store.clientID}}))
GROUP BY taskTypes.team, Applications.reviewOrder
ORDER BY taskTypes.team, Applications.reviewOrder;
