SELECT 
    Tasks.taskID, Tasks.appID, Tasks.startDate, Tasks.completedDate, Tasks.dueDate, 
    Tasks.taskTypeID, Tasks.assignedTo, Tasks.taskNotes, Tasks.filePath, 
    Applications.appID, Applications.portalAppID, Applications.applicationName, 
    taskTypes.taskName, Tasks.taskStatus, taskTypes.team, taskStatusTypes.sortOrder
FROM taskStatusTypes 
INNER JOIN (
    (Applications 
    INNER JOIN Entities ON Applications.entityID = Entities.entityID) 
    INNER JOIN Tasks ON Applications.appID = Tasks.appID) 
INNER JOIN taskTypes ON Tasks.taskTypeID = taskTypes.taskTypeID 
ON taskStatusTypes.taskStatus = Tasks.taskStatus
			
WHERE Entities.clientID = {{appsmith.store.clientID}}
    AND ((Tasks.appID= '{{Select6.selectedOptionValue}}'))
    AND ({{Switch2.isSwitchedOn}} OR Tasks.taskStatus <> "Complete")
    AND ({{!Select10.selectedOptionValue}} OR (Applications.applicationType = '{{Select10.selectedOptionValue}}'))
    AND ({{!Select7.selectedOptionValue}} OR (taskTypes.team = '{{Select7.selectedOptionValue}}'))
ORDER BY sortOrder ASC, startDate DESC;
