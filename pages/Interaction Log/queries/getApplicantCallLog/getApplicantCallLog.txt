
SELECT 
    i.spokeWith,
    i.inReferenceTo,
    i.commsDate,
    i.commsType,
    i.commsSubject,
    i.importance,
    i.notes,
    i.CTCstaff,
    i.clientID,
		i.entityID,
    e.entityName,
    a.applicationName,
		a.portalAppID,
		a.applicationType
FROM interactionLog i
LEFT JOIN Entities e ON i.entityID = e.entityID 
LEFT JOIN Applications a ON i.applicationID = a.appID
	WHERE i.entityID = {{entitySelect.selectedOptionValue}}
  AND archived = "No"
	ORDER BY commsDate DESC
;
