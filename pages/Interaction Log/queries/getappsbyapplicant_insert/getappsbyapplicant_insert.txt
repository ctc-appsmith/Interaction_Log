SELECT Applications.applicationName, Applications.portalAppID, Applications.appID, Entities.entityName, Clients.clientID
FROM (Applications INNER JOIN Entities ON Applications.entityID = Entities.entityID) INNER JOIN Clients ON Entities.clientID = Clients.clientID
WHERE (((Entities.entityName)={{insert_form.formData.inReferenceTo}}) AND ((Clients.clientID)={{appsmith.store.clientID}}))
ORDER BY Applications.applicationName;
